# Frontend Integration

go-auth ships a TypeScript API client and React hooks in the companion `frontend-next` template. This guide documents the patterns.

## API Client

The API client is at `src/lib/api/`. It's organized by domain:

| File | Exports |
|---|---|
| `client.ts` | `apiClient` — fetch wrapper with cookie credentials |
| `auth.ts` | Login, register, logout, me |
| `sessions.ts` | List, revoke sessions |
| `invites.ts` | Invite info, register with invite |
| `orgs.ts` | All org CRUD + invite acceptance |

### Usage

```typescript
import { authApi } from '@/lib/api/auth';
import { orgsApi } from '@/lib/api/orgs';

// Login
const { user, session } = await authApi.login({
    email: 'test@example.com',
    password: 'Passw0rd!',
});

// Get current user
const me = await authApi.me();

// List user's orgs
const orgs = await orgsApi.list();

// Accept org invite
await orgsApi.acceptInvite('invite-code-123');
```

### Client Configuration

```typescript
// src/lib/api/client.ts
const API_BASE = process.env.NEXT_PUBLIC_API_URL || 'http://localhost:8080';

export const apiClient = {
    async fetch<T>(path: string, options?: RequestInit): Promise<T> {
        const res = await fetch(`${API_BASE}${path}`, {
            credentials: 'include', // sends cookies
            headers: {
                'Content-Type': 'application/json',
                ...options?.headers,
            },
            ...options,
        });
        if (!res.ok) throw await res.json();
        return res.json();
    },
};
```

## React Hooks

Hooks are in `src/lib/hooks/`:

| Hook | File | Returns |
|---|---|---|
| `useAuth` | `use-auth.tsx` | `{ user, session, login, logout, loading, error }` |
| `useOrg` | `use-org.ts` | `{ org, members, invites, loading }` |
| `useOrgList` | `use-org-list.ts` | `{ orgs, loading, createOrg }` |

### Auth Provider

Wrap your app:

```tsx
// app/providers.tsx
import { AuthProvider } from '@/lib/hooks/use-auth';

export default function RootLayout({ children }) {
    return (
        <AuthProvider>
            {children}
        </AuthProvider>
    );
}
```

### Using Auth

```tsx
import { useAuth } from '@/lib/hooks/use-auth';

function ProfilePage() {
    const { user, session, logout, loading } = useAuth();

    if (loading) return <div>Loading...</div>;
    if (!user) return <div>Please log in</div>;

    return (
        <div>
            <h1>{user.name}</h1>
            <p>{user.email}</p>
            <button onClick={logout}>Logout</button>
        </div>
    );
}
```

## CSRF Handling

When CSRF is enabled, the client must:

1. GET `/auth/csrf-token` to get the CSRF cookie
2. Read the cookie value
3. Send it in the `X-CSRF-Token` header on state-changing requests

The API client handles this automatically:

```typescript
async fetchWithCSRF<T>(path: string, options?: RequestInit): Promise<T> {
    // Get CSRF token from cookie
    const csrfToken = getCookie('csrf_token');
    return apiClient.fetch<T>(path, {
        ...options,
        headers: {
            'X-CSRF-Token': csrfToken,
            ...options?.headers,
        },
    });
}

function getCookie(name: string): string | undefined {
    const match = document.cookie.match(new RegExp(`(^| )${name}=([^;]+)`));
    return match?.[2];
}
```

## Organization Frontend

### Accepting an Invite

```tsx
// app/invite/page.tsx
'use client';
import { useSearchParams, useRouter } from 'next/navigation';
import { orgsApi } from '@/lib/api/orgs';

export default function InvitePage() {
    const searchParams = useSearchParams();
    const router = useRouter();
    const code = searchParams.get('code');

    const handleAccept = async () => {
        try {
            await orgsApi.acceptInvite(code!);
            router.push('/dashboard/orgs');
        } catch (err) {
            // handle error
        }
    };

    return <button onClick={handleAccept}>Accept Invite</button>;
}
```

### Org Role Checking

```tsx
// Check org role on the client
function canManageMembers(member: OrgMemberDetail): boolean {
    return member.role === 'owner' || member.role === 'admin';
}
```

## TypeScript Types

Defined in `src/lib/api/types.ts`:

```typescript
interface User {
    id: string;
    email: string;
    name: string;
    role: 'user' | 'admin';
    isVerified: boolean;
    isBanned: boolean;
    createdAt: string;
    updatedAt: string;
}

interface Organization {
    id: string;
    name: string;
    slug: string;
    created_by?: string;
    owner_count: number;
    member_count: number;
    created_at: string;
    updated_at: string;
}

interface OrgMemberDetail {
    org_id: string;
    user_id: string;
    role: 'owner' | 'admin' | 'member';
    joined_at: string;
    user: User;
}

interface OrgInvite {
    id: string;
    org_id: string;
    email: string;
    role: 'owner' | 'admin' | 'member';
    rawCode?: string;
    invited_by: string;
    expires_at: string;
    created_at: string;
}
```
