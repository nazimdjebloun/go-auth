## Service Input Types

```go
type RegisterInput struct {
    Email, Password, Name string
}

type LoginInput struct {
    Email, Password, IP, UserAgent string
}

type ForgotPasswordInput struct {
    Email string
}

type ResetPasswordInput struct {
    Code, NewPassword string
}

type ChangePasswordInput struct {
    UserID, OldPassword, NewPassword string
}

type ConfirmSetPasswordInput struct {
    UserID, Code, NewPassword string
}

type ConfirmDeleteAccountInput struct {
    UserID, Code string
}

type CreateInviteInput struct {
    Email string
}

type CompleteInviteInput struct {
    Code, Name, Password, ConfirmPassword string
}

type CompleteInviteResult struct {
    User                     *domain.User
    Session                  *domain.Session
    SessionToken, RefreshToken string
}

type ListInvitesInput struct {
    Offset, Limit int
    Search, Status string
}

type AdminListUsersInput struct {
    Offset, Limit int
    Email, Search, Role string
    OrderBy, OrderDirection string
}

type AdminListUsersResult struct {
    Users []domain.User
    Total int
}

type AdminListUserSessionsInput struct {
    UserID string
    Offset, Limit int
}

type CreateUserInput struct {
    Email, Password, Name, Role string
}

type CreateOrgInput struct {
    Name, Slug string
    CreatedBy string
}

type UpdateOrgInput struct {
    OrgID, Name, Slug string
}

type AddMemberInput struct {
    OrgID, UserID string
    Role domain.OrgRole
}

type UpdateMemberRoleInput struct {
    OrgID, UserID string
    Role domain.OrgRole
}

type CreateOrgInviteInput struct {
    OrgID, Email string
    Role domain.OrgRole
}

type AcceptInviteInput struct {
    UserID, Code string
}
```
